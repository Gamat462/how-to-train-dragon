import re
import pandas as pd


def ke_angka(nilai: object) -> int | None:
    if nilai is None or (isinstance(nilai, float) and pd.isna(nilai)):
        return None
    if isinstance(nilai, (int,)):
        return int(nilai)
    teks = re.sub(r"[^0-9\-]", "", str(nilai))
    return int(teks) if teks not in ("", "-") else None

Alat ukur baru: puncak memori, bukan cuma waktu

LAKUKAN Sesi Python baru. tracemalloc itu bawaan Python, tidak perlu dipasang.
import pandas as pd, time, tracemalloc
BERKAS = "transaksi.csv"

def puncak(f):
    """Jalankan f, kembalikan (detik, puncak MB, hasil)."""
    tracemalloc.start()
    t = time.perf_counter(); hasil = f(); dt = time.perf_counter() - t
    _, puncak_b = tracemalloc.get_traced_memory(); tracemalloc.stop()
    return dt, puncak_b/1e6, hasil

def bertahap(chunk=100_000):
    total = {}
    for bagian in pd.read_csv(BERKAS, usecols=["cabang","nilai"],
                              dtype={"cabang":"category","nilai":"int32"},
                              chunksize=chunk):
        for k, v in bagian.groupby("cabang", observed=True)["nilai"].sum().items():
            total[k] = total.get(k, 0) + v
    return pd.Series(total).sort_index()
HARUS MUNCUL

import csv

def baris_besar(path, ambang):
    """Keluarkan baris satu per satu; jangan pernah simpan semuanya."""
    with open(path, newline="") as f:
        for baris in csv.DictReader(f):
            if int(baris["nilai"]) > ambang:

import pandas as pd


def omzet_per_cabang(df: pd.DataFrame) -> pd.DataFrame:
    """Ringkas omzet dan jumlah transaksi per cabang, terbesar dulu."""
    return (df.groupby("cabang", observed=True)
              .agg(transaksi=("nilai", "size"), omzet=("nilai", "sum"))
              .sort_values("omzet", ascending=False)
              .reset_index())
